
#include <Wire.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_BNO055.h>
#include <utility/imumaths.h>

// Khai bao chan I2C
#define SDA_PIN 4
#define SCL_PIN 5

// Khoi tao BNO055
Adafruit_BNO055 bno = Adafruit_BNO055(55, 0x28);

void setup() {
  Serial.begin(115200);

  Wire.begin(SDA_PIN, SCL_PIN);

  Serial.println("Starting BNO055...");

  if (!bno.begin()) {
    Serial.println("ERROR: BNO055 not detected!");
    while (1) {
      delay(100);
    }
  }

  // Su dung tinh the noi bo
  bno.setExtCrystalUse(true);

  Serial.println("BNO055 READY!");
  delay(1000);
}

void loop() {

  // Doc goc Euler
  imu::Vector<3> euler =
    bno.getVector(Adafruit_BNO055::VECTOR_EULER);

  float yaw   = euler.x();
  float roll  = euler.y();
  float pitch = euler.z();

  // Doc gia toc
  imu::Vector<3> accel =
    bno.getVector(Adafruit_BNO055::VECTOR_ACCELEROMETER);

  // Doc van toc goc
  imu::Vector<3> gyro =
    bno.getVector(Adafruit_BNO055::VECTOR_GYROSCOPE);

  Serial.println("========== BNO055 ==========");

  Serial.print("Roll  : ");
  Serial.print(roll);
  Serial.println(" deg");

  Serial.print("Pitch : ");
  Serial.print(pitch);
  Serial.println(" deg");

  Serial.print("Yaw   : ");
  Serial.print(yaw);
  Serial.println(" deg");

  Serial.println("----------------------------");

  Serial.print("Accel X: ");
  Serial.print(accel.x());
  Serial.print(" | Y: ");
  Serial.print(accel.y());
  Serial.print(" | Z: ");
  Serial.println(accel.z());

  Serial.print("Gyro X: ");
  Serial.print(gyro.x() * 57.2958);
  Serial.print(" | Y: ");
  Serial.print(gyro.y() * 57.2958);
  Serial.print(" | Z: ");
  Serial.println(gyro.z() * 57.2958);

  // Trang thai hieu chuan
  uint8_t sys, gyroCal, accelCal, magCal;
  bno.getCalibration(&sys, &gyroCal, &accelCal, &magCal);

  Serial.print("Calibration: ");
  Serial.print(sys);
  Serial.print(" ");
  Serial.print(gyroCal);
  Serial.print(" ");
  Serial.print(accelCal);
  Serial.print(" ");
  Serial.println(magCal);

  Serial.println("============================\n");

  delay(200);
}
